Option Explicit
'===============================================================================
' 週報作成マクロ
'
'  「年間行事予定（月別のワークシート）」ブックの ４月～３月 シートから、
'  指定した週の月曜日から31日分の
'     日直（C列）・校内行事（D列）・校外行事（E列）・提出文書・備考（F列）
'  を読み取り、週報シート（「原本」のコピー）の A4:E10（月～日の行事欄）と
'  D11（今後の予定）、E2（週の期間）に直接書き込みます。
'  （F～L 列の貼付データ G13:L43 は使いません。）
'  月をまたぐ週でも自動で次の月のシートから読み取ります。
'
'  使い方：Alt+F8 →「週報作成」→ 実行
'    ① 週の番号を入力（記入済みの最後の週の次が入っています）
'    ② 週の開始日（月曜日）を確認。前の週＋7日が入っているので、
'       長期休業明けなどで違うときだけ直す
'  （ショートカットキーは Alt+F8 →［オプション］で設定できます）
'===============================================================================

'----- 設定 -----------------------------------------------------------------
Private Const SHEET_GENPON As String = "原本"     ' 週報のひな形シート
Private Const SHEET_EXTRA As String = "追加情報"  ' 校内行事を編集するキーワード表（A列：キーワード、B列：編集後）
Private Const DAYS_TO_COPY As Long = 31           ' 読み取る日数（月曜日から31日分）
Private Const WEEK_DAYS As Long = 7               ' 週報の行事欄にする日数（月～日）
Private Const SRC_FIRST_ROW As Long = 3           ' 月シートで「1日」がある行
Private Const TIDY_SCHEDULE As Boolean = True     ' 今後の予定欄(D11)で行事のない日を省く
Private Const NAME_SRC_PATH As String = "週報作成_行事予定パス"
Private Const NAME_PENDING As String = "週報作成_予約"      ' 後半の処理に渡す週番号・開始日
Private Const NAME_NOTE As String = "週報作成_注記"         ' 完了メッセージに添える注記
Private Const MAX_ROW_HEIGHT As Double = 409          ' Excel の行の高さの上限（ポイント）
Private Const MIN_FONT_SIZE As Double = 6            ' 縮小するときのフォントの下限
Private Const HEIGHT_MARGIN As Double = 1.1          ' 測った高さに持たせる余裕（画面と印刷の差の分）

Private quietMode As Boolean     ' テスト用：True のときメッセージを出さない
Private lastMessage As String

'===============================================================================
' メイン：週番号を聞いて週報シートを作る
'===============================================================================
Public Sub 週報作成()
    ' 1) 週番号（既定値＝記入済みの最後の週の次）
    Dim defWeek As Long
    defWeek = LastFilledWeek() + 1

    Dim ans As String, weekNo As Long
    ans = InputBox("作成する週の番号を入力してください。", "週報作成", CStr(defWeek))
    If ans = "" Then Exit Sub
    ans = StrConv(ans, vbNarrow)
    If Not IsNumeric(ans) Then
        MsgBox "週の番号は数字で入力してください。", vbExclamation, "週報作成"
        Exit Sub
    End If
    weekNo = CLng(ans)
    If weekNo < 1 Then Exit Sub

    ' 2) 週の開始日（月曜日）：前の週から計算して表示し、訂正できるようにする
    Dim startDate As Date, note As String
    If Not AskStartDate(weekNo, startDate, note) Then Exit Sub

    ' 3) 行事予定ブックがすでに開いていれば、そのまま作る
    Dim wbSrc As Workbook
    Set wbSrc = FindOpenSource()
    If Not wbSrc Is Nothing Then
        FinishReport wbSrc, weekNo, startDate, note, False
        Exit Sub
    End If

    ' 4) 開いていなければファイルを開く
    '    Ctrl+Shift のショートカットで実行すると、Excel はファイルを開いた直後に
    '    マクロを止めてしまうことがある。そこで、続きの処理（ContinueWeeklyReport）を
    '    先に予約してから開く。止まっても止まらなくても、続きは予約から1回だけ動く。
    Dim path As String
    path = ChooseSourcePath()
    If path = "" Then Exit Sub
    SaveSourcePath path
    SaveText NAME_PENDING, weekNo & "|" & CLng(startDate)
    SaveText NAME_NOTE, Replace(note, vbLf, "\n")
    Application.OnTime Now, "'" & ThisWorkbook.Name & "'!ContinueWeeklyReport"
    Application.ScreenUpdating = False
    Workbooks.Open Filename:=path, UpdateLinks:=0, ReadOnly:=True
End Sub

' 後半：開いた行事予定ブックから週報を作る（週報作成 から予約して呼ばれる）
Public Sub ContinueWeeklyReport()
    Dim s As String, p As Variant
    s = LoadText(NAME_PENDING)
    If s = "" Then Exit Sub                  ' 予約がなければ何もしない
    SaveText NAME_PENDING, ""
    p = Split(s, "|")
    If UBound(p) < 1 Then Exit Sub

    Dim weekNo As Long, startDate As Date, note As String, path As String
    weekNo = CLng(p(0))
    startDate = CDate(CLng(p(1)))
    note = Replace(LoadText(NAME_NOTE), "\n", vbLf)
    SaveText NAME_NOTE, ""
    path = LoadSavedPath()

    Dim wb As Workbook, wbSrc As Workbook
    For Each wb In Application.Workbooks
        If StrComp(wb.FullName, path, vbTextCompare) = 0 Then Set wbSrc = wb
    Next wb
    Application.ScreenUpdating = True
    If wbSrc Is Nothing Then
        ShowMessage "行事予定ブックを開けませんでした。" & vbLf & path, vbExclamation
        SaveSourcePath ""
        Exit Sub
    End If
    If Not IsSourceWorkbook(wbSrc) Then
        ShowMessage "選んだファイルに「４月」～「３月」のシートが見つかりません。" & vbLf & wbSrc.Name, vbExclamation
        wbSrc.Close SaveChanges:=False
        SaveSourcePath ""
        Exit Sub
    End If
    FinishReport wbSrc, weekNo, startDate, note, True
End Sub

Private Sub FinishReport(ByVal wbSrc As Workbook, ByVal weekNo As Long, ByVal startDate As Date, _
                         ByVal note As String, ByVal closeSrc As Boolean)
    ThisWorkbook.Activate
    Dim msg As String
    msg = MakeWeeklyReport(wbSrc, weekNo, startDate, True)
    If closeSrc Then wbSrc.Close SaveChanges:=False
    ThisWorkbook.Activate
    Application.ScreenUpdating = True
    If msg <> "" Then ShowMessage msg & note, vbInformation
End Sub

Private Sub ShowMessage(ByVal msg As String, ByVal style As Long)
    lastMessage = msg
    If Not quietMode Then MsgBox msg, style, "週報作成"
End Sub

' 開始日を前の週から計算して表示し、訂正を受け付ける
'   訂正なし（OK のみ）→ 計算した日付のまま
'   訂正あり           → 入力した日付（月曜日でなければ、その週の月曜日）
Private Function AskStartDate(ByVal weekNo As Long, ByRef result As Date, ByRef note As String) As Boolean
    Dim calc As Date, prevNo As Long, prompt As String, defText As String
    note = ""
    If PrevWeekStart(weekNo, calc, prevNo) Then
        defText = Format(calc, "yyyy/m/d")
        prompt = "第" & weekNo & "週の開始日（月曜日）" & vbLf & vbLf & _
                 "前の週（第" & prevNo & "週：" & Format(calc - 7 * (weekNo - prevNo), "m/d") & _
                 "～）から計算すると " & Format(calc, "m/d") & "（" & WeekdayChar(calc) & "）です。" & vbLf & vbLf & _
                 "長期休業などで違う場合は、正しい日付に直してください。" & vbLf & _
                 "（例：2026/8/31　または　8/31）"
    Else
        defText = ""
        prompt = "第" & weekNo & "週の開始日（月曜日）を入力してください。" & vbLf & _
                 "（前の週のシートから日付を求められませんでした）" & vbLf & vbLf & _
                 "（例：2026/8/31　または　8/31）"
    End If

    Dim s As String
    s = InputBox(prompt, "週報作成：週の開始日", defText)
    If s = "" Then Exit Function                         ' キャンセル
    s = StrConv(Trim$(s), vbNarrow)

    If defText <> "" And s = defText Then
        result = calc                                    ' 訂正なし
    Else
        If Not IsDate(s) Then
            MsgBox "日付として読み取れませんでした：" & s, vbExclamation, "週報作成"
            Exit Function
        End If
        Dim d As Date
        d = DateValue(s)
        If InStr(s, "/") = InStrRev(s, "/") And defText <> "" Then
            ' 「8/31」のように年を省略したときは、計算した日付に近い年を使う
            d = DateSerial(Year(calc), Month(d), Day(d))
            If d - calc > 180 Then d = DateSerial(Year(calc) - 1, Month(d), Day(d))
            If calc - d > 180 Then d = DateSerial(Year(calc) + 1, Month(d), Day(d))
        End If
        result = MondayOf(d)
        If result <> d Then
            note = vbLf & vbLf & "※入力された " & Format(d, "m/d") & "（" & WeekdayChar(d) & _
                   "）は月曜日ではないため、その週の月曜日 " & Format(result, "m/d") & " にしました。"
        End If
    End If
    AskStartDate = True
End Function

'===============================================================================
' 本体：週報シートを用意してデータを書き込む（戻り値＝完了メッセージ、中止時は ""）
'===============================================================================
Public Function MakeWeeklyReport(ByVal wbSrc As Workbook, ByVal weekNo As Long, _
                                 ByVal startDate As Date, ByVal askOverwrite As Boolean) As String
    Dim ws As Worksheet
    Set ws = FindSheet(ThisWorkbook, CStr(weekNo))

    If ws Is Nothing Then
        Dim wsGen As Worksheet
        Set wsGen = FindSheet(ThisWorkbook, SHEET_GENPON)
        If wsGen Is Nothing Then
            MsgBox "「" & SHEET_GENPON & "」シートが見つかりません。", vbCritical, "週報作成"
            Exit Function
        End If
        wsGen.Copy After:=LastWeekSheet()
        Set ws = ActiveSheet
        ws.Name = CStr(weekNo)
        ws.Visible = xlSheetVisible
    ElseIf askOverwrite Then
        If IsFilledSheet(ws) Then
            If MsgBox("「" & ws.Name & "」シートにはすでにデータがあります。" & vbLf & _
                      "行事予定から読み直して上書きしますか？" & vbLf & _
                      "（A4:E10・D11・E2 と、お知らせ欄の「行事・教育課程の反省」の一覧を書き換えます）", _
                      vbYesNo + vbQuestion, "週報作成") <> vbYes Then Exit Function
        End If
    End If

    Application.ScreenUpdating = False

    ' 年度・月・週の初日
    ws.Range("F12").Value = SchoolYear(startDate)
    ws.Range("H12").Value = Month(startDate)
    ws.Range("F13").ClearContents
    ws.Range("F13").Value = startDate

    ' 31日分を行事予定ブックから読み取る（週シートの F～L 列は使わない）
    Dim i As Long, d As Date, wsM As Worksheet, curMonth As Long, sr As Long
    Dim missing As String
    Dim dayNo(0 To DAYS_TO_COPY - 1) As Long, wdChar(0 To DAYS_TO_COPY - 1) As String
    Dim duty(0 To DAYS_TO_COPY - 1) As String, inside(0 To DAYS_TO_COPY - 1) As String
    Dim outside(0 To DAYS_TO_COPY - 1) As String, remark(0 To DAYS_TO_COPY - 1) As String
    curMonth = 0
    For i = 0 To DAYS_TO_COPY - 1
        d = startDate + i
        If Month(d) <> curMonth Then
            curMonth = Month(d)
            Set wsM = MonthSheet(wbSrc, curMonth)
            If wsM Is Nothing Then missing = missing & " " & curMonth & "月"
        End If
        dayNo(i) = Day(d)
        wdChar(i) = WeekdayChar(d)
        If Not wsM Is Nothing Then
            sr = SRC_FIRST_ROW + Day(d) - 1
            duty(i) = CleanText(wsM.Cells(sr, "C"))                                  ' 日直
            inside(i) = TidySeparators(CleanText(wsM.Cells(sr, "D")))                ' 校内行事
            outside(i) = BreakBeforeDetail(CleanText(wsM.Cells(sr, "E")))            ' 校外行事
            remark(i) = CleanText(wsM.Cells(sr, "F"))                                ' 提出文書・備考
        End If
    Next i

    ' 「追加情報」シートのキーワードで校内行事を編集する（シートがなければ何もしない）
    Dim nReplaced As Long, weekInside(0 To WEEK_DAYS - 1) As String
    nReplaced = ApplyExtraInfoToArray(inside, WEEK_DAYS)
    For i = 0 To WEEK_DAYS - 1
        weekInside(i) = inside(i)
    Next i

    ' 週報の各欄に直接書き込む
    WriteWeekCells ws, startDate, duty, inside, outside, remark
    WriteUpcoming ws, dayNo, wdChar, inside
    WriteReflectionList ws, weekInside    ' A11 の反省欄にその週の主な行事を書き出す
    FitRowHeights ws

    Application.ScreenUpdating = True
    ws.Activate
    ws.Range("A1").Select

    MakeWeeklyReport = "第" & weekNo & "週（" & Format(startDate, "m/d") & "～" & _
                       Format(startDate + 6, "m/d") & "）の週報を作成しました。" & vbLf & _
                       "行事予定ブック：" & wbSrc.Name
    If nReplaced > 0 Then
        MakeWeeklyReport = MakeWeeklyReport & vbLf & _
            "「" & SHEET_EXTRA & "」で校内行事を " & nReplaced & " か所編集しました。"
    End If
    If missing <> "" Then
        MakeWeeklyReport = MakeWeeklyReport & vbLf & vbLf & _
            "※次の月のシートが行事予定ブックに見つからず、空欄にしています：" & missing
    End If
End Function

'===============================================================================
' 補助
'===============================================================================

' 前の週の開始日＋7日を返す（見つかった前の週の番号も返す）
'   さかのぼって最初に正しい開始日を持つ週を使う（空のひな形シートは飛ばす）
Private Function PrevWeekStart(ByVal weekNo As Long, ByRef result As Date, ByRef prevNo As Long) As Boolean
    Dim k As Long, ws As Worksheet, d As Date
    For k = weekNo - 1 To 1 Step -1
        Set ws = FindSheet(ThisWorkbook, CStr(k))
        If Not ws Is Nothing Then
            If SheetStartDate(ws, d) Then
                result = d + 7 * (weekNo - k)
                prevNo = k
                PrevWeekStart = True
                Exit Function
            End If
        End If
    Next k
End Function

' 週シートの開始日（月曜日）。古い様式の「週初日」(H2/G2) → F13 の順に探す
'   月曜日で、今年度の日付であるものだけを正しい開始日とみなす
Private Function SheetStartDate(ByVal ws As Worksheet, ByRef result As Date) As Boolean
    Dim addr As Variant, v As Variant
    For Each addr In Array("H2", "G2", "F13")
        v = ws.Range(addr).Value
        If Not IsError(v) Then
            ' 日付、または日付を表す数値（2000/1/1～2099/12/31）だけを対象にする
            If VarType(v) = vbDate Or ((VarType(v) = vbDouble Or VarType(v) = vbLong _
                                       Or VarType(v) = vbInteger) And v >= 36526 And v <= 73415) Then
                If Weekday(CDate(v), vbMonday) = 1 And IsThisSchoolYear(CDate(v)) Then
                    result = CDate(v)
                    SheetStartDate = True
                    Exit Function
                End If
            End If
        End If
    Next addr
End Function

' 原本 F12 の年度と同じ年度の日付か（F12 が読めなければ判定しない）
Private Function IsThisSchoolYear(ByVal d As Date) As Boolean
    Dim ws As Worksheet, y As Variant
    IsThisSchoolYear = True
    Set ws = FindSheet(ThisWorkbook, SHEET_GENPON)
    If ws Is Nothing Then Exit Function
    y = ws.Range("F12").Value
    If IsNumeric(y) And Not IsEmpty(y) Then IsThisSchoolYear = (SchoolYear(d) = CLng(y))
End Function

' 記入済みの週シートのうち、番号が最大のもの
Private Function LastFilledWeek() As Long
    Dim ws As Worksheet, n As Long
    For Each ws In ThisWorkbook.Worksheets
        If IsNumeric(ws.Name) Then
            n = CLng(ws.Name)
            If n > LastFilledWeek Then
                If IsFilledSheet(ws) Then LastFilledWeek = n
            End If
        End If
    Next ws
End Function

' 週の内容が入っているシートか
'   ・F13 に日付が値として入っている（このマクロで作った週）
'   ・または、以前の様式で G13:L43 に貼付データがある
'   原本のひな形は F13 が数式なので、空とみなす
Private Function IsFilledSheet(ByVal ws As Worksheet) As Boolean
    Dim c As Range
    Set c = ws.Range("F13")
    If Not c.HasFormula Then
        If VarType(c.Value) = vbDate Then IsFilledSheet = True: Exit Function
    End If
    IsFilledSheet = (Application.WorksheetFunction.CountA(ws.Range("G13:L43")) > 0)
End Function

Private Function MondayOf(ByVal d As Date) As Date
    MondayOf = d - (Weekday(d, vbMonday) - 1)
End Function

Private Function WeekdayChar(ByVal d As Date) As String
    WeekdayChar = Mid("日月火水木金土", Weekday(d, vbSunday), 1)
End Function

' 開いているブックの中から行事予定ブックを探す
Private Function FindOpenSource() As Workbook
    Dim wb As Workbook
    For Each wb In Application.Workbooks
        If Not wb Is ThisWorkbook Then
            If IsSourceWorkbook(wb) Then
                Set FindOpenSource = wb
                Exit Function
            End If
        End If
    Next wb
End Function

' 行事予定ファイルの場所：前回のパス（ファイルが残っていれば）→ なければ選んでもらう
Private Function ChooseSourcePath() As String
    Dim path As String
    path = LoadSavedPath()
    If path <> "" Then
        If Dir(path) = "" Then path = ""
    End If
    If path = "" Then
        Dim f As Variant
        f = Application.GetOpenFilename("Excel ファイル (*.xlsx;*.xlsm;*.xls),*.xlsx;*.xlsm;*.xls", , _
                                        "年間行事予定（月別のワークシート）のファイルを選んでください")
        If VarType(f) = vbBoolean Then Exit Function
        path = CStr(f)
    End If
    ChooseSourcePath = path
End Function

Private Function IsSourceWorkbook(ByVal wb As Workbook) As Boolean
    IsSourceWorkbook = Not (MonthSheet(wb, 4) Is Nothing) And Not (MonthSheet(wb, 12) Is Nothing)
End Function

' 月シート（「１２月」のような全角、または「12月」）を返す
Private Function MonthSheet(ByVal wb As Workbook, ByVal m As Long) As Worksheet
    Set MonthSheet = FindSheet(wb, StrConv(CStr(m), vbWide) & "月")
    If MonthSheet Is Nothing Then Set MonthSheet = FindSheet(wb, CStr(m) & "月")
End Function

Private Function FindSheet(ByVal wb As Workbook, ByVal sheetName As String) As Worksheet
    Dim ws As Worksheet
    For Each ws In wb.Worksheets
        If ws.Name = sheetName Then
            Set FindSheet = ws
            Exit Function
        End If
    Next ws
End Function

' 数字名の週シートのうち番号が最大のもの（なければ最後のシート）
Private Function LastWeekSheet() As Worksheet
    Dim ws As Worksheet, best As Worksheet, n As Long, bestN As Long
    bestN = -1
    For Each ws In ThisWorkbook.Worksheets
        If IsNumeric(ws.Name) Then
            n = CLng(ws.Name)
            If n > bestN Then bestN = n: Set best = ws
        End If
    Next ws
    If best Is Nothing Then Set best = ThisWorkbook.Worksheets(ThisWorkbook.Worksheets.Count)
    Set LastWeekSheet = best
End Function

' 空の数式が返す 0 やエラー値を空欄にする
Private Function CleanText(ByVal c As Range) As String
    Dim v As Variant
    v = c.Value
    If IsError(v) Or IsEmpty(v) Then Exit Function
    If VarType(v) = vbString Then
        If Trim$(v) = "0" Then Exit Function
        CleanText = v
    ElseIf IsNumeric(v) Then
        If v = 0 Then Exit Function
        CleanText = CStr(v)
    Else
        CleanText = CStr(v)
    End If
End Function

Private Function SchoolYear(ByVal d As Date) As Long
    If Month(d) < 4 Then SchoolYear = Year(d) - 1 Else SchoolYear = Year(d)
End Function

' 今後の予定（8日目以降）の1日分：「 7(月)　〇行事　〇行事」の形（複数の行事は改行で区切る）
'   日付が1桁なら前に半角スペースを入れて、2桁の日付と「(」の位置をそろえる
'   行事は「〇」を付けて改行で区切る。「＊」で始まる行事には「〇」を付けない
Private Function ScheduleLine(ByVal dayNo As Long, ByVal wd As String, ByVal inside As String) As String
    Dim fw As String
    fw = ChrW(&H3000)                        ' 全角スペース
    If inside = "" Then Exit Function
    ScheduleLine = IIf(dayNo < 10, " ", "") & dayNo & "(" & wd & ")" & fw & ScheduleText(inside)
End Function

' 今後の予定用：校内行事の文字列の各行事に「〇」を付けて改行で区切る
'   ・「＊」で始まる行事には「〇」を付けない
'   ・行事名の途中の「＊」は、前に全角スペースを入れる（クラブ③＊… → クラブ③　＊…）
'   （先頭に改行を1つ付けて「行頭の＊」と「途中の＊」を区別し、最後に取り除く）
Private Function ScheduleText(ByVal t As String) As String
    Dim fw As String, x As String
    fw = ChrW(&H3000)
    x = Replace(t, " ＊", "＊")                                  ' 「＊」の前の半角スペースを取る
    x = vbLf & "〇" & Replace(x, fw, vbLf & "〇")                ' 行事ごとに改行＋〇
    x = Replace(x, "〇" & vbLf, "〇")                            ' 「〇」の直後の改行は取る（〇冬季休業…）
    Do While InStr(x, "〇〇") > 0
        x = Replace(x, "〇〇", "〇")
    Loop
    x = Replace(x, "〇＊", "＊")                                 ' 「＊」で始まる行事は〇なし
    x = Replace(x, "＊", fw & "＊")                              ' 「＊」の前に全角スペース
    x = Replace(x, vbLf & fw & "＊", vbLf & "＊")                ' 行頭の「＊」は元に戻す
    ScheduleText = Mid$(x, 2)                                    ' 先頭の改行を取る
End Function

' 同じ日の2行目以降の字下げ：日付の直後の「〇」の位置にそろえる
'   「日付(曜)＋全角スペース」は半角スペース8つ分の幅（日付が1桁の日は前に半角スペースを足してある）
'   2行目以降は、そこからさらに半角スペース2つ分下げる
'   ・行頭が「〇」「＊」の行：半角スペース10つ（8＋2）
'   ・「〇」「＊」が付かない行（行事名の途中の改行）：さらに「〇」1文字分（半角スペース2つ）下げて12
Private Function IndentContinuation(ByVal t As String) As String
    Const BASE As Long = 10
    Dim lines As Variant, i As Long, ch As String
    lines = Split(t, vbLf)
    For i = LBound(lines) + 1 To UBound(lines)
        ch = Left$(lines(i), 1)
        If ch = "〇" Or ch = "＊" Then
            lines(i) = Space$(BASE) & lines(i)
        Else
            lines(i) = Space$(BASE + 2) & lines(i)
        End If
    Next i
    IndentContinuation = Join(lines, vbLf)
End Function

' D11（今後の予定）：月曜日から8日目以降の行事を書き込む
'   1行につなげる日（8～18日目）は、改行を全角スペースにして「〇A　〇B」のように区切る
Private Sub WriteUpcoming(ByVal ws As Worksheet, ByRef dayNo() As Long, ByRef wdChar() As String, _
                          ByRef inside() As String)
    Dim i As Long, fw As String, ln As String, out As String
    fw = ChrW(&H3000)
    For i = WEEK_DAYS To DAYS_TO_COPY - 1
        If TIDY_SCHEDULE Then
            ln = ScheduleLine(dayNo(i), wdChar(i), inside(i))
            If ln <> "" Then
                If i <= WEEK_DAYS + 10 Then
                    ln = Replace(ln, vbLf, fw)
                Else
                    ln = IndentContinuation(ln)
                End If
                If out <> "" Then out = out & vbLf
                out = out & ln
            End If
        Else
            ' 原本の元の表示：行事のない日も書き、行事ごとの改行は取る
            ln = dayNo(i) & "(" & wdChar(i) & ")" & fw & "〇" & Replace(inside(i), fw, vbLf & "〇")
            If out <> "" Then out = out & vbLf
            out = out & Replace(ln, vbLf, "")
        End If
    Next i
    ws.Range("D11").Value = out
End Sub

' 括弧の中の全角スペースは、行事の区切りではないので、同じ幅の別の空白（EM SPACE）にする
'   例：冬季休業（12/24 ～　1/5) → 「〇1/5)」と分けて書き出さない
'   括弧は「（」「(」で開き、「）」「)」で閉じる（閉じていない括弧は何もしない）
Private Function ProtectBracketSpaces(ByVal t As String) As String
    Dim i As Long, j As Long, depth As Long, ch As String, inner As String
    Dim fw As String, em As String
    fw = ChrW(&H3000): em = ChrW(&H2003)
    If InStr(t, fw) = 0 Then ProtectBracketSpaces = t: Exit Function
    i = 1
    Do While i <= Len(t)
        ch = Mid$(t, i, 1)
        If ch = "（" Or ch = "(" Then
            depth = 0
            For j = i To Len(t)
                ch = Mid$(t, j, 1)
                If ch = "（" Or ch = "(" Then depth = depth + 1
                If ch = "）" Or ch = ")" Then depth = depth - 1
                If depth = 0 Then Exit For
            Next j
            If j <= Len(t) Then
                inner = Mid$(t, i, j - i + 1)
                t = Left$(t, i - 1) & Replace(inner, fw, em) & Mid$(t, j + 1)
                i = j
            End If
        End If
        i = i + 1
    Loop
    ProtectBracketSpaces = t
End Function

' 校内行事の区切り（全角スペース）を整える：続いたものは1つに、先頭・末尾のものと改行は取る
Private Function TidySeparators(ByVal t As String) As String
    Dim fw As String
    fw = ChrW(&H3000)
    t = ProtectBracketSpaces(t)
    Do While InStr(t, fw & fw) > 0
        t = Replace(t, fw & fw, fw)
    Loop
    t = Replace(t, fw & vbLf, vbLf)
    t = Replace(t, vbLf & fw, vbLf)
    Do While Len(t) > 0 And (Left$(t, 1) = fw Or Left$(t, 1) = vbLf)
        t = Mid$(t, 2)
    Loop
    Do While Len(t) > 0 And (Right$(t, 1) = fw Or Right$(t, 1) = vbLf)
        t = Left$(t, Len(t) - 1)
    Loop
    TidySeparators = t
End Function

' 校内行事の文字列を「〇行事」「＊補足」の1行ずつに分ける（末尾の改行はなし）
'   ・全角スペース（行事の区切り）を「改行＋〇」に、「＊」を「改行＋＊」にする
'   ・「＊」の前の半角・全角スペースは取ってから改行する
'   ・「〇」だけの行（「＊」で始まる補足の前など）と空行を取り除く
Private Function InsideText(ByVal t As String) As String
    Dim fw As String, x As String
    fw = ChrW(&H3000)                        ' 全角スペース（行事の区切り）
    x = Replace(t, " ＊", "＊")
    x = Replace(x, fw & "＊", "＊")
    x = Replace(x, "＊", vbLf & "＊")
    x = Replace(x, fw, vbLf & "〇")
    x = "〇" & x
    x = Replace(x, "〇" & vbLf, "〇")                            ' 「〇」の直後の改行は取る（〇冬季休業…）
    Do While InStr(x, "〇〇") > 0
        x = Replace(x, "〇〇", "〇")
    Loop
    x = Replace(x, "〇＊", "＊")                                 ' 「＊」で始まる補足には〇を付けない
    x = Replace(x, vbLf & vbLf, vbLf)
    InsideText = x
End Function

' 事前入力内容（G4:K10）の1セルを文字列で返す（空・0・エラーは ""）
Private Function PreEntry(ByVal ws As Worksheet, ByVal col As String, ByVal r As Long) As String
    PreEntry = CleanText(ws.Range(col & r))
End Function

' 週報の A4:E10 と E2 に、1週間（月～日）の内容を直接書き込む
'   A：月日＋日直、B：曜日、C：校内行事、D：校外行事、E：備考（報告等）
'   事前入力内容（I4:K10）があれば、C・D・E の末尾に足す（○・〆付き）
'   土日の行（9・10行目）は、校内行事・校外行事・備考のうち値のあるものを改行でつないで C 列にまとめ、
'   C:E を結合して左寄せにする
Private Sub WriteWeekCells(ByVal ws As Worksheet, ByVal startDate As Date, ByRef duty() As String, _
                           ByRef inside() As String, ByRef outside() As String, ByRef remark() As String)
    Dim i As Long, r As Long, d As Date, fw As String, s As String, hk As Long
    Dim preIn As String, preOut As String, preRem As String
    Dim hName() As String, hDate() As String, hMean() As String, nHol As Long
    fw = ChrW(&H3000)
    nHol = LoadHolidays(hName, hDate, hMean)

    ' A 列の元の文字色（原本のもの）
    Dim wsGen As Worksheet, baseColor As Long
    baseColor = ws.Range("A4").Font.Color
    Set wsGen = FindSheet(ThisWorkbook, SHEET_GENPON)
    If Not wsGen Is Nothing Then baseColor = wsGen.Range("A4").Font.Color

    ' 週の期間（月～金）
    ws.Range("E2").Value = Month(startDate) & "月" & Day(startDate) & "日～" & _
                           Month(startDate + 4) & "月" & Day(startDate + 4) & "日"

    For i = 0 To WEEK_DAYS - 1
        r = 4 + i
        d = startDate + i
        preIn = PreEntry(ws, "I", r)
        preOut = PreEntry(ws, "J", r)
        preRem = PreEntry(ws, "K", r)

        ws.Range("A" & r).Value = Month(d) & "/" & Day(d) & vbLf & duty(i)
        ' 日直が校外行事の「（　）」内にもいるとき（部分一致を含む）は、A 列の文字を赤にする
        If DutyInBrackets(duty(i), outside(i) & vbLf & preOut) Then
            ws.Range("A" & r).Font.Color = RGB(255, 0, 0)
        Else
            ws.Range("A" & r).Font.Color = baseColor
        End If
        ws.Range("B" & r).Value = "(" & WeekdayChar(d) & ")"

        ' 「国民の祝日」シートの祝日名が校内行事にあるか（部分一致を含む）
        hk = 0
        If nHol > 0 Then hk = FindHoliday(inside(i), hName, nHol)

        If i <= 4 And hk = 0 Then            ' 月～金（祝日でない日）：C・D・E を別々に書く
            With ws.Range("C" & r & ":E" & r)
                If .MergeCells Then
                    .UnMerge                 ' 前に祝日で結合していたときは、結合をやめて元の配置に戻す
                    If Not wsGen Is Nothing Then
                        ws.Range("C" & r).HorizontalAlignment = wsGen.Range("C" & r).HorizontalAlignment
                        ws.Range("C" & r).VerticalAlignment = wsGen.Range("C" & r).VerticalAlignment
                        ws.Range("D" & r).HorizontalAlignment = wsGen.Range("D" & r).HorizontalAlignment
                        ws.Range("D" & r).VerticalAlignment = wsGen.Range("D" & r).VerticalAlignment
                        ws.Range("E" & r).HorizontalAlignment = wsGen.Range("E" & r).HorizontalAlignment
                        ws.Range("E" & r).VerticalAlignment = wsGen.Range("E" & r).VerticalAlignment
                    End If
                End If
            End With
            s = ""
            If inside(i) <> "" Then s = InsideText(inside(i)) & vbLf
            If preIn <> "" Then s = s & "○" & preIn
            ws.Range("C" & r).Value = s
            ws.Range("C" & r).Font.Strikethrough = False

            s = ""
            If outside(i) <> "" Then s = outside(i) & vbLf
            If preOut <> "" Then s = s & preOut
            ws.Range("D" & r).Value = s

            s = ""
            If remark(i) <> "" Then s = RemarkText(remark(i)) & vbLf
            If preRem <> "" Then s = s & RemarkText(preRem)
            ws.Range("E" & r).Value = s
            StrikeDoneLines ws.Range("E" & r)
        Else                                 ' 土日と祝日：C:E を結合して1つにまとめる
            s = ""
            Dim rest As String
            rest = inside(i)
            If hk > 0 Then
                ' 祝日名＋改行＋日付＋意味（行事のほかの内容は、その下に続ける）
                s = hName(hk) & vbLf & hDate(hk) & "『" & hMean(hk) & "』"
                rest = RemoveHolidayItems(inside(i), hName(hk))
            End If
            If rest <> "" Then s = JoinLine(s, InsideText(rest))
            If preIn <> "" Then s = JoinLine(s, "○" & preIn)
            s = JoinLine(s, outside(i))
            s = JoinLine(s, preOut)
            If remark(i) <> "" Then s = JoinLine(s, RemarkText(remark(i)))
            If preRem <> "" Then s = JoinLine(s, RemarkText(preRem))
            With ws.Range("C" & r & ":E" & r)
                If .MergeCells Then .UnMerge
                ws.Range("D" & r & ":E" & r).ClearContents
                ws.Range("C" & r).Value = s
                ws.Range("C" & r).Font.Strikethrough = False
                StrikeDoneLines ws.Range("C" & r)
                .Merge
                .HorizontalAlignment = xlLeft
                .VerticalAlignment = xlTop
                .WrapText = True
            End With
        End If
    Next i
End Sub

' 備考の文字列：複数あるときは、改行した内容の先頭にも「〆」を付ける（空の行は省く）
'   全角スペースの前にも「〆」を付ける（従来どおり）
Private Function RemarkText(ByVal t As String) As String
    Dim lines As Variant, i As Long, ln As String, out As String, fw As String
    fw = ChrW(&H3000)
    lines = Split(Replace(t, vbCrLf, vbLf), vbLf)
    For i = LBound(lines) To UBound(lines)
        ln = lines(i)
        If Trim$(Replace(ln, fw, "")) <> "" Then
            If Left$(ln, 1) <> "〆" Then ln = "〆" & ln
            ln = Left$(ln, 1) & Replace(Mid$(ln, 2), fw, "〆" & fw)
            If out <> "" Then out = out & vbLf
            out = out & ln
        End If
    Next i
    RemarkText = out
End Function

' 備考のうち「済」「済み」で終わる行（〆で始まる行）に取り消し線を付ける（見え消し）
'   書き込み直すときは、まず取り消し線を外してからやり直す
Private Sub StrikeDoneLines(ByVal c As Range)
    Dim txt As String, lines As Variant, i As Long, pos As Long, ln As String, t As String
    On Error Resume Next                     ' 見た目だけなので、失敗しても続ける
    c.Font.Strikethrough = False
    txt = CStr(c.Value)
    lines = Split(txt, vbLf)
    pos = 1
    For i = LBound(lines) To UBound(lines)
        ln = lines(i)
        t = Trim$(Replace(ln, ChrW(&H3000), " "))
        If Left$(ln, 1) = "〆" And (Right$(t, 1) = "済" Or Right$(t, 2) = "済み") Then
            c.Characters(pos, Len(ln)).Font.Strikethrough = True
        End If
        pos = pos + Len(ln) + 1
    Next i
    On Error GoTo 0
End Sub

' 日直（職員名）が、校外行事の「（…）」「(…)」の中に出てくるか（部分一致を含む）
'   括弧の中を「、」「・」「，」「：」「／」やスペースで区切り、区切りごとに日直の名前と
'   一方がもう一方を含んでいれば一致とする（2文字以上。例：「杉山」と「杉山先生」）
Private Function DutyInBrackets(ByVal duty As String, ByVal outsideText As String) As Boolean
    Dim name As String, i As Long, j As Long, depth As Long, ch As String, inner As String
    Dim tokens As Variant, k As Long, tk As String
    name = NormName(duty)
    If Len(name) < 2 Then Exit Function
    i = 1
    Do While i <= Len(outsideText)
        ch = Mid$(outsideText, i, 1)
        If ch = "（" Or ch = "(" Then
            depth = 0
            For j = i To Len(outsideText)
                ch = Mid$(outsideText, j, 1)
                If ch = "（" Or ch = "(" Then depth = depth + 1
                If ch = "）" Or ch = ")" Then depth = depth - 1
                If depth = 0 Then Exit For
            Next j
            If j > Len(outsideText) Then j = Len(outsideText)       ' 閉じていなければ最後まで
            inner = Mid$(outsideText, i + 1, j - i - 1)
            inner = Replace(inner, "、", " "): inner = Replace(inner, "，", " ")
            inner = Replace(inner, "・", " "): inner = Replace(inner, "：", " ")
            inner = Replace(inner, ":", " "): inner = Replace(inner, "／", " ")
            inner = Replace(inner, "/", " "): inner = Replace(inner, ChrW(&H3000), " ")
            inner = Replace(inner, ChrW(&H2003), " ")
            tokens = Split(inner, " ")
            For k = LBound(tokens) To UBound(tokens)
                tk = NormName(CStr(tokens(k)))
                If Len(tk) >= 2 Then
                    If InStr(tk, name) > 0 Or InStr(name, tk) > 0 Then DutyInBrackets = True: Exit Function
                End If
            Next k
            i = j
        End If
        i = i + 1
    Loop
End Function

' 名前の比較用：空白と異体字セレクタを取り、全角を半角にそろえる
Private Function NormName(ByVal t As String) As String
    Dim i As Long, c As Long, out As String
    t = Replace(t, ChrW(&H3000), "")
    t = Replace(t, " ", "")
    t = Replace(t, vbLf, "")
    For i = 1 To Len(t)
        c = AscW(Mid$(t, i, 1))
        If c < 0 Then c = c + 65536
        If c >= &HFE00& And c <= &HFE0F& Then
            ' 異体字セレクタ（VS1～16）は飛ばす
        ElseIf c = &HDB40& Then
            i = i + 1                        ' 異体字セレクタ（VS17～256）はサロゲートペアなので2文字飛ばす
        Else
            out = out & Mid$(t, i, 1)
        End If
    Next i
    NormName = StrConv(out, vbNarrow)
End Function

' 「国民の祝日」シート（A 祝日名、B 日付、C 意味。2行目から）を読む（戻り値＝件数）
'   日付が日付の値なら「11月23日」の形に、「1月の第2月曜日」のような文字ならそのまま使う
Private Function LoadHolidays(ByRef nm() As String, ByRef dt() As String, ByRef mn() As String) As Long
    Dim wsH As Worksheet, r As Long, lastRow As Long, n As Long, v As Variant
    Set wsH = FindSheet(ThisWorkbook, "国民の祝日")
    If wsH Is Nothing Then Exit Function
    lastRow = wsH.Cells(wsH.Rows.Count, "A").End(xlUp).Row
    If lastRow < 2 Then Exit Function
    ReDim nm(1 To lastRow): ReDim dt(1 To lastRow): ReDim mn(1 To lastRow)
    For r = 2 To lastRow
        If Trim$(CStr(wsH.Cells(r, "A").Value)) <> "" Then
            n = n + 1
            nm(n) = Trim$(CStr(wsH.Cells(r, "A").Value))
            v = wsH.Cells(r, "B").Value
            If IsError(v) Then
                dt(n) = ""
            ElseIf VarType(v) = vbDate Then
                dt(n) = Month(v) & "月" & Day(v) & "日"
            ElseIf IsNumeric(v) And Not IsEmpty(v) Then
                If v > 20000 And v < 80000 Then dt(n) = Month(CDate(v)) & "月" & Day(CDate(v)) & "日" Else dt(n) = CStr(v)
            Else
                dt(n) = CStr(v)
            End If
            If IsError(wsH.Cells(r, "C").Value) Then mn(n) = "" Else mn(n) = CStr(wsH.Cells(r, "C").Value)
        End If
    Next r
    LoadHolidays = n
End Function

' 校内行事の1つの行事が、祝日名に当てはまるか（どちらかがもう一方を含めば一致。「＊」で始まる補足は対象外）
Private Function IsHolidayItem(ByVal item As String, ByVal holName As String) As Boolean
    Dim a As String, b As String
    If Left$(Trim$(item), 1) = "＊" Then Exit Function
    a = NormForMatch(item): b = NormForMatch(holName)
    If a = "" Or b = "" Then Exit Function
    If InStr(a, b) > 0 Then IsHolidayItem = True: Exit Function
    If Len(a) >= 2 Then IsHolidayItem = (InStr(b, a) > 0)
End Function

' その日の校内行事に当てはまる祝日の番号（なければ 0）
Private Function FindHoliday(ByVal inside As String, ByRef nm() As String, ByVal n As Long) As Long
    Dim items As Variant, i As Long, k As Long
    If inside = "" Then Exit Function
    items = Split(Replace(inside, vbLf, ChrW(&H3000)), ChrW(&H3000))
    For i = LBound(items) To UBound(items)
        For k = 1 To n
            If IsHolidayItem(CStr(items(i)), nm(k)) Then FindHoliday = k: Exit Function
        Next k
    Next i
End Function

' 校内行事から、祝日に当てはまる行事を取り除く
Private Function RemoveHolidayItems(ByVal inside As String, ByVal holName As String) As String
    Dim items As Variant, i As Long, out As String
    items = Split(inside, ChrW(&H3000))
    For i = LBound(items) To UBound(items)
        If Not IsHolidayItem(CStr(items(i)), holName) Then
            If out <> "" Then out = out & ChrW(&H3000)
            out = out & items(i)
        End If
    Next i
    RemoveHolidayItems = out
End Function

' 空でない文字列だけを改行でつなぐ
Private Function JoinLine(ByVal a As String, ByVal b As String) As String
    If b = "" Then JoinLine = a: Exit Function
    If a = "" Then JoinLine = b Else JoinLine = a & vbLf & b
End Function

' A11「※行事・教育課程の反省＞＞＞」の下に、その週の主な校内行事を「〇」付きで1行ずつ書き出す
'   ・その週（月～日）の校内行事（inside）を全角スペース・改行で区切って1つずつ取り出す
'   ・「＊」で始まる補足、祝日（「国民の祝日」シートの名前と「振替休日」）、重複は書き出さない
'   ・行事名の途中の「＊」以降（補足）は省く
'   ・見出しの下にある空白だけの行（と、前に書き出した行）を置き換える。ほかの部分はそのまま
Private Sub WriteReflectionList(ByVal ws As Worksheet, ByRef inside() As String)
    Const HEADING As String = "※行事・教育課程の反省"
    Const PREFIX As String = "　〇"          ' 行頭（字下げ＋〇）
    Dim txt As String, lines As Variant, i As Long, hi As Long, j As Long
    txt = CStr(ws.Range("A11").Value)
    If txt = "" Then Exit Sub
    txt = Replace(txt, vbCrLf, vbLf)
    lines = Split(txt, vbLf)
    hi = -1
    For i = LBound(lines) To UBound(lines)
        If Left$(lines(i), Len(HEADING)) = HEADING Then hi = i: Exit For
    Next i
    If hi < 0 Then Exit Sub                  ' 見出しがなければ何もしない

    ' 見出しの下の、空白だけの行と前に書き出した行を飛ばした位置 j
    j = hi + 1
    Do While j <= UBound(lines)
        If Trim$(Replace(lines(j), ChrW(&H3000), "")) <> "" And Left$(lines(j), Len(PREFIX)) <> PREFIX Then Exit Do
        j = j + 1
    Loop

    ' その週の主な校内行事
    Dim holidays As String, r As Long, parts As Variant, k As Long, it As String
    Dim listText As String, seen As String
    holidays = HolidayNames()
    Dim excl() As String, nExcl As Long
    nExcl = LoadExcludeWords(excl)
    For r = LBound(inside) To UBound(inside)
        parts = Split(Replace(inside(r), vbLf, ChrW(&H3000)), ChrW(&H3000))
        For k = LBound(parts) To UBound(parts)
            it = parts(k)
            If InStr(2, it, "＊") > 0 Then it = Left$(it, InStr(2, it, "＊") - 1)   ' 「＊」以降の補足は省く
            it = Trim$(it)
            If it <> "" And it <> "0" Then
                If Left$(it, 1) <> "＊" And Left$(it, 1) <> "*" _
                   And InStr(holidays, vbTab & it & vbTab) = 0 _
                   And InStr(seen, vbTab & it & vbTab) = 0 _
                   And Not IsExcludedEvent(it, excl, nExcl) Then
                    seen = seen & vbTab & it & vbTab
                    listText = listText & PREFIX & it & vbLf
                End If
            End If
        Next k
    Next r

    ' 組み立て直す：見出しまで ＋ 行事一覧 ＋ 空行 ＋ 残り
    '   残す行の書式（太字・サイズ・色など）を覚えておき、書き込んだあとに付け直す
    '   （.Value で書き込むと「【特別日課】」の太字などの部分的な書式が消えるため）
    Dim n As Long, oldPos() As Long, newPos() As Long, keep() As Boolean, pos As Long
    n = UBound(lines)
    ReDim oldPos(0 To n): ReDim newPos(0 To n): ReDim keep(0 To n)
    pos = 1
    For i = 0 To n
        oldPos(i) = pos
        pos = pos + Len(lines(i)) + 1
    Next i

    Dim out As String
    For i = LBound(lines) To hi
        newPos(i) = Len(out) + 1: keep(i) = True
        out = out & lines(i) & vbLf
    Next i
    out = out & listText
    If j <= UBound(lines) Then
        out = out & vbLf                     ' 次の見出しとの間を1行あける
        For i = j To UBound(lines)
            newPos(i) = Len(out) + 1: keep(i) = True
            out = out & lines(i)
            If i < UBound(lines) Then out = out & vbLf
        Next i
    End If

    ' 残す行の、先頭の文字の書式を記録
    Dim fB() As Variant, fI() As Variant, fS() As Variant, fC() As Variant, fN() As Variant, fU() As Variant
    ReDim fB(0 To n): ReDim fI(0 To n): ReDim fS(0 To n): ReDim fC(0 To n): ReDim fN(0 To n): ReDim fU(0 To n)
    On Error Resume Next                     ' 書式の付け直しは見た目だけなので、失敗しても続ける
    For i = 0 To n
        If keep(i) And Len(lines(i)) > 0 Then
            With ws.Range("A11").Characters(oldPos(i), 1).Font
                fB(i) = .Bold: fI(i) = .Italic: fS(i) = .Size: fC(i) = .Color: fN(i) = .Name: fU(i) = .Underline
            End With
        End If
    Next i
    On Error GoTo 0

    ws.Range("A11").Value = out

    ' 書式を付け直す（行事一覧の行は、書き込んだときの基本の書式のまま）
    On Error Resume Next
    For i = 0 To n
        If keep(i) And Len(lines(i)) > 0 Then
            With ws.Range("A11").Characters(newPos(i), Len(lines(i))).Font
                .Bold = fB(i): .Italic = fI(i): .Size = fS(i): .Color = fC(i): .Name = fN(i): .Underline = fU(i)
            End With
        End If
    Next i
    On Error GoTo 0
End Sub

' 行事名に除外の言葉を含むか
Private Function IsExcludedEvent(ByVal it As String, ByRef words() As String, ByVal n As Long) As Boolean
    Dim i As Long
    Dim t As String, parts As Variant, k As Long, p As Long, pos As Long, ok As Boolean, w As String
    t = NormForMatch(it)
    For i = 1 To n
        ' 「＊」は任意の文字列。分けた部分が順に含まれていれば一致（部分一致）
        w = NormForMatch(words(i))
        If Len(Replace(w, "*", "")) > 0 Then
            parts = Split(w, "*")
            pos = 1: ok = True
            For k = LBound(parts) To UBound(parts)
                If parts(k) <> "" Then
                    p = InStr(pos, t, parts(k))
                    If p = 0 Then ok = False: Exit For
                    pos = p + Len(parts(k))
                End If
            Next k
            If ok Then IsExcludedEvent = True: Exit Function
        End If
    Next i
End Function

' 比べるための文字列：全角を半角に、大文字を小文字にして、空白（全角・EM・半角・改行）を取る
Private Function NormForMatch(ByVal t As String) As String
    t = StrConv(t, vbNarrow)
    t = Replace(t, ChrW(&H3000), "")
    t = Replace(t, ChrW(&H2003), "")
    t = Replace(t, " ", "")
    t = Replace(t, vbTab, "")
    t = Replace(t, vbCr, "")
    t = Replace(t, vbLf, "")
    NormForMatch = LCase$(t)
End Function

' 校外行事：詳細を表す「（…）」の前で改行する（各行の最初の該当する括弧だけ）
'   例：第１３回共同実施（１３：３０　新島小　池林）
'    → 第１３回共同実施
'      　（１３：３０　新島小　池林）
'   詳細の括弧＝中に全角スペース・「：」・「:」・「、」のどれかを含むもの
'   （「（他校研修）」「（佐原区）」のような名前の一部の括弧では改行しない）
Private Function BreakBeforeDetail(ByVal t As String) As String
    Const INDENT As String = "　"           ' 改行後の行頭に入れる文字（不要なら "" にする）
    Dim lines As Variant, i As Long, ln As String, p As Long, q As Long
    Dim depth As Long, inner As String, head As String
    If t = "" Then Exit Function
    lines = Split(t, vbLf)
    For i = LBound(lines) To UBound(lines)
        ln = lines(i)
        p = InStr(2, ln, "（")
        Do While p > 0
            ' 対応する閉じ括弧を探す（入れ子の括弧を数える）
            depth = 0
            For q = p To Len(ln)
                If Mid$(ln, q, 1) = "（" Then depth = depth + 1
                If Mid$(ln, q, 1) = "）" Then depth = depth - 1
                If depth = 0 Then Exit For
            Next q
            If q > Len(ln) Then Exit Do                  ' 閉じ括弧がない
            inner = Mid$(ln, p + 1, q - p - 1)
            If InStr(inner, ChrW(&H3000)) > 0 Or InStr(inner, "：") > 0 Or _
               InStr(inner, ":") > 0 Or InStr(inner, "、") > 0 Then
                head = Left$(ln, p - 1)
                Do While Len(head) > 0 And (Right$(head, 1) = ChrW(&H3000) Or Right$(head, 1) = " ")
                    head = Left$(head, Len(head) - 1)
                Loop
                If head <> "" Then ln = head & vbLf & INDENT & Mid$(ln, p)
                Exit Do
            End If
            p = InStr(q + 1, ln, "（")
        Loop
        lines(i) = ln
    Next i
    BreakBeforeDetail = Join(lines, vbLf)
End Function

' 追加情報：校内行事に特定のキーワードがあれば、情報を追加・置き換える
'
'  「追加情報」シート（1行目は見出し、2行目から）
'     A列 キーワード   例：ALT＊
'     B列 編集後       例：ALT＊（②３年③４年④６年⑤５年）
'  ・キーワードの「＊」は番号（数字・丸数字）に当てはまり、編集後の「＊」に同じ番号を入れる
'       ALT⑥ → ALT⑥（②３年③４年④６年⑤５年）
'  ・「＊」がないキーワードは、そのまま編集後の文字列に置き換える
'  ・キーワードは行事名の先頭から一致したものだけが対象（「代表委員会」は「委員会＊」に一致しない）
'  ・すでに編集済みのものは二重に付け足さない
'  ・「編集後」が空欄のキーワードは何もしない
'  ・「編集後」の中の全角スペースは、行事の区切り（〇に置き換わる）と区別するため
'    同じ幅の別の空白文字（EM SPACE）にして書き込む → 〇が付かずにそのまま表示される
'  ・編集するのはその週（月～日）の校内行事だけ。今後の予定（D11）は編集しない
'  ・追加情報反映（単独実行）は、週シートの C4:C10 の文字を直接置き換える
'===============================================================================

' 表示中の週シートに、追加情報をもう一度反映する（追加情報シートを直したあとなど）
Public Sub 追加情報反映()
    If FindSheet(ThisWorkbook, SHEET_EXTRA) Is Nothing Then
        CreateExtraInfoSheet
        MsgBox "「" & SHEET_EXTRA & "」シートを作りました。" & vbLf & _
               "A列にキーワード、B列に編集後の文字列を入力してから、もう一度実行してください。", _
               vbInformation, "週報作成"
        Exit Sub
    End If
    If Not IsNumeric(ActiveSheet.Name) Then
        MsgBox "週のシート（名前が番号のシート）を表示してから実行してください。", vbExclamation, "週報作成"
        Exit Sub
    End If
    Dim ws As Worksheet, n As Long
    Set ws = ActiveSheet
    Application.ScreenUpdating = False
    n = ApplyExtraInfoToSheet(ws)
    Dim inside(0 To WEEK_DAYS - 1) As String
    ReadInsideFromSheet ws, inside
    WriteReflectionList ws, inside
    FitRowHeights ws
    Application.ScreenUpdating = True
    MsgBox "校内行事を " & n & " か所編集しました。", vbInformation, "週報作成"
End Sub

' 追加情報のキーワード表を読む（戻り値＝有効なキーワードの数）
Private Function LoadExtraRules(ByRef keys() As String, ByRef edits() As String) As Long
    Dim wsX As Worksheet
    Set wsX = FindSheet(ThisWorkbook, SHEET_EXTRA)
    If wsX Is Nothing Then Exit Function

    Dim lastRow As Long, r As Long, nRule As Long, k As String
    lastRow = wsX.UsedRange.Row + wsX.UsedRange.Rows.Count - 1
    If lastRow < 1 Then Exit Function
    ReDim keys(1 To lastRow): ReDim edits(1 To lastRow)
    For r = 1 To lastRow
        k = Trim$(CStr(wsX.Cells(r, "A").Value))
        If k <> "" And k <> "キーワード" And CStr(wsX.Cells(r, "B").Value) <> "" Then
            nRule = nRule + 1
            keys(nRule) = k
            edits(nRule) = CStr(wsX.Cells(r, "B").Value)
        End If
    Next r
    LoadExtraRules = nRule
End Function

' 校内行事の文字列の配列（先頭から nDays 日分）に追加情報を反映し、編集した数を返す
'   今後の予定（8日目以降）は編集しない
Private Function ApplyExtraInfoToArray(ByRef inside() As String, ByVal nDays As Long) As Long
    Dim keys() As String, edits() As String, nRule As Long
    nRule = LoadExtraRules(keys, edits)
    If nRule = 0 Then Exit Function

    Dim i As Long, j As Long, cnt As Long, total As Long, t As String
    For i = 0 To nDays - 1
        t = inside(i)
        If t <> "" Then
            cnt = 0
            For j = 1 To nRule
                t = ApplyOneRule(t, keys(j), edits(j), cnt)
            Next j
            If cnt > 0 Then
                inside(i) = t
                total = total + cnt
            End If
        End If
    Next i
    ApplyExtraInfoToArray = total
End Function

' 週シートの C4:C10 の文字に追加情報を反映し、編集した数を返す（追加情報反映 から使う）
Private Function ApplyExtraInfoToSheet(ByVal ws As Worksheet) As Long
    Dim keys() As String, edits() As String, nRule As Long
    nRule = LoadExtraRules(keys, edits)
    If nRule = 0 Then Exit Function

    Dim r As Long, j As Long, cnt As Long, total As Long, t As String, v As Variant
    For r = 4 To 10
        v = ws.Range("C" & r).Value
        If Not IsError(v) Then
            t = CStr(v)
            If t <> "" Then
                cnt = 0
                For j = 1 To nRule
                    t = ApplyOneRule(t, keys(j), edits(j), cnt)
                Next j
                If cnt > 0 Then
                    ws.Range("C" & r).Value = t
                    total = total + cnt
                End If
            End If
        End If
    Next r
    ApplyExtraInfoToSheet = total
End Function

' 週シートの C4:C10 から、校内行事（行頭が「〇」の行）を取り出す（反省欄の書き出し用）
'   行事の区切りは全角スペースにする。行事名の中の全角スペースは同じ幅の別の空白にして区切りと区別する
Private Sub ReadInsideFromSheet(ByVal ws As Worksheet, ByRef inside() As String)
    Dim r As Long, lines As Variant, i As Long, out As String, v As Variant
    For r = 4 To 10
        v = ws.Range("C" & r).Value
        out = ""
        If Not IsError(v) Then
            lines = Split(Replace(CStr(v), vbCrLf, vbLf), vbLf)
            For i = LBound(lines) To UBound(lines)
                If Left$(lines(i), 1) = "〇" Then
                    If out <> "" Then out = out & ChrW(&H3000)
                    out = out & Replace(Mid$(lines(i), 2), ChrW(&H3000), ChrW(&H2003))
                End If
            Next i
        End If
        inside(r - 4) = out
    Next r
End Sub

' 1つのキーワードを文字列 t に当てはめる（編集した数を cnt に足す）
Public Function ApplyOneRule(ByVal t As String, ByVal keyWord As String, ByVal edited As String, _
                             ByRef cnt As Long) As String
    Const STAR As String = "＊"
    keyWord = Replace(keyWord, "*", STAR)
    edited = Replace(edited, "*", STAR)
    If edited = "" Then ApplyOneRule = t: Exit Function     ' 編集後が空欄なら何もしない
    ' 編集後の中の全角スペースは EM SPACE（同じ幅の空白）にして、区切りとして扱われないようにする
    edited = Replace(edited, ChrW(&H3000), ChrW(&H2003))

    Dim wild As Boolean, kPre As String, kPost As String, ePre As String, ePost As String
    Dim sp As Long
    sp = InStr(keyWord, STAR)
    wild = (sp > 0)
    If wild Then
        kPre = Left$(keyWord, sp - 1)
        kPost = Mid$(keyWord, sp + 1)
    Else
        kPre = keyWord
    End If
    ApplyOneRule = t
    If kPre = "" Then Exit Function                ' 「＊」で始まるキーワードは扱わない

    sp = InStr(edited, STAR)
    If wild And sp > 0 Then
        ePre = Left$(edited, sp - 1)
        ePost = Mid$(edited, sp + 1)
    End If

    Dim pos As Long, p As Long, j As Long, endPos As Long, ok As Boolean
    Dim cap As String, rep As String
    pos = 1
    Do
        p = InStr(pos, t, kPre)
        If p = 0 Then Exit Do
        ok = IsLeftBoundary(t, p)
        j = p + Len(kPre)
        cap = ""
        If ok And wild Then
            Do While j <= Len(t)
                If Not IsNumberChar(Mid$(t, j, 1)) Then Exit Do
                j = j + 1
            Loop
            cap = Mid$(t, p + Len(kPre), j - p - Len(kPre))
            If cap = "" Then ok = False                    ' 番号がなければ対象外
            If ok And kPost <> "" Then
                If Mid$(t, j, Len(kPost)) = kPost Then j = j + Len(kPost) Else ok = False
            End If
        End If
        endPos = j
        If ok Then ok = IsRightBoundary(t, endPos)

        If ok Then
            If wild And sp > 0 Then rep = ePre & cap & ePost Else rep = edited
            If Mid$(t, p, Len(rep)) = rep Then
                pos = p + Len(rep)                         ' すでに編集済み
            Else
                t = Left$(t, p - 1) & rep & Mid$(t, endPos)
                cnt = cnt + 1
                pos = p + Len(rep)
            End If
        Else
            pos = p + 1
        End If
    Loop
    ApplyOneRule = t
End Function

' キーワードの前が行事の区切り（先頭・スペース・改行・〇・＊・、）か
Private Function IsLeftBoundary(ByVal t As String, ByVal p As Long) As Boolean
    If p <= 1 Then IsLeftBoundary = True: Exit Function
    IsLeftBoundary = InStr(ChrW(&H3000) & " " & vbLf & vbCr & "〇○＊*、，,・", Mid$(t, p - 1, 1)) > 0
End Function

' キーワードの後ろが行事の区切り（末尾・スペース・改行・〇・括弧・＊・・・、）か
Private Function IsRightBoundary(ByVal t As String, ByVal p As Long) As Boolean
    If p > Len(t) Then IsRightBoundary = True: Exit Function
    IsRightBoundary = InStr(ChrW(&H3000) & " " & vbLf & vbCr & "〇○（(＊*・、，,", Mid$(t, p, 1)) > 0
End Function

' 番号の文字か（半角・全角数字、丸数字 ①～⑳ と 21～50 の丸数字、白抜き・黒丸の数字）
Private Function IsNumberChar(ByVal ch As String) As Boolean
    Dim c As Long
    c = AscW(ch)
    If c < 0 Then c = c + 65536
    Select Case c
        Case &H30& To &H39&, &HFF10& To &HFF19&, &H2460& To &H2473&, &H3251& To &H325F&, _
             &H32B1& To &H32BF&, &H2776& To &H277F&, &H2780& To &H2789&
            IsNumberChar = True
    End Select
End Function

' C4:E8（A・B 列も含む）、土日の C9・C10、D11 の内容がすべて見えるように行の高さを調整する。
'   ・行の高さ＝その行のセルが全部表示できる高さの最大値（原本の高さより低くはしない）
'   ・上限 409 ポイントでも入りきらないセルは、フォントを 0.5 ずつ小さくする
Private Sub FitRowHeights(ByVal ws As Worksheet)
    Dim wsGen As Worksheet, tmp As Worksheet, r As Long
    Set wsGen = FindSheet(ThisWorkbook, SHEET_GENPON)
    Application.Calculate                ' 書き込んだデータで数式を計算し終えてから測る

    ' 高さを測るための一時シート（最後に削除）
    Set tmp = ThisWorkbook.Worksheets.Add(After:=ThisWorkbook.Worksheets(ThisWorkbook.Worksheets.Count))
    For r = 4 To 8
        FitOneRow ws, wsGen, tmp, r, Array("A", "B", "C", "D", "E")
    Next r
    For r = 9 To 10                          ' 土日（C:E は結合セル）
        FitOneRow ws, wsGen, tmp, r, Array("A", "B", "C")
    Next r
    FitOneRow ws, wsGen, tmp, 11, Array("A", "D")

    Application.DisplayAlerts = False
    tmp.Delete
    Application.DisplayAlerts = True
    ws.Activate

    ' 行の高さが変わっても画像（QR コードなど）が原本と同じセルの位置に来るようにする
    If Not wsGen Is Nothing Then AlignShapesToGenpon ws, wsGen
End Sub

' 原本の図形と同じ名前の図形を、原本と同じセルからの位置へ置き直す
'   （原本の画像は「セルに合わせて移動しない」設定のため、行が高くなると下の行に重なる）
Private Sub AlignShapesToGenpon(ByVal ws As Worksheet, ByVal wsGen As Worksheet)
    Dim gs As Shape, s As Shape, r As Long, offset As Double
    On Error Resume Next                     ' 見た目だけの調整なので、失敗しても作成は続ける
    For Each gs In wsGen.Shapes
        Set s = Nothing
        Set s = ws.Shapes(gs.Name)
        If Not s Is Nothing Then
            r = RowAtTop(wsGen, gs.Top)              ' 原本で図形の上端がある行
            offset = gs.Top - wsGen.Rows(r).Top
            s.Top = ws.Rows(r).Top + offset
        End If
    Next gs
    On Error GoTo 0
End Sub

' 上端の位置 y（ポイント）が含まれる行の番号
Private Function RowAtTop(ByVal ws As Worksheet, ByVal y As Double) As Long
    Dim r As Long
    r = 1
    Do While r < 1000
        If ws.Rows(r + 1).Top > y Then Exit Do
        r = r + 1
    Loop
    RowAtTop = r
End Function

Private Sub FitOneRow(ByVal ws As Worksheet, ByVal wsGen As Worksheet, ByVal tmp As Worksheet, _
                      ByVal r As Long, ByVal cols As Variant)
    Dim need As Double, h As Double, c As Variant
    If wsGen Is Nothing Then need = ws.Rows(r).RowHeight Else need = wsGen.Rows(r).RowHeight
    For Each c In cols
        h = FitOneCell(ws.Range(c & r), wsGen, tmp)
        If h > need Then need = h
    Next c
    If need > MAX_ROW_HEIGHT Then need = MAX_ROW_HEIGHT
    ws.Rows(r).RowHeight = need
End Sub

' セル（結合セルなら結合範囲全体）を全部表示するのに必要な高さを返す。
' 上限を超えるときはフォントを小さくして収める。
Private Function FitOneCell(ByVal cell As Range, ByVal wsGen As Worksheet, ByVal tmp As Worksheet) As Double
    Dim area As Range, topCell As Range, v As Variant, txt As String
    Set area = cell.MergeArea
    Set topCell = area.Cells(1, 1)

    ' フォントサイズは原本のサイズから測り直す（作り直したときに小さいままにならないように）
    ' ただし、文字ごとにサイズなどが違うセル（A11 のお知らせ欄など）は、Font.Size が Null になる。
    ' そのセルは先頭の文字のサイズで測り、部分的な書式を崩さないようにフォントは変えない。
    Dim size As Double, fs As Variant, mixed As Boolean
    fs = topCell.Font.Size
    If Not wsGen Is Nothing Then fs = wsGen.Range(topCell.Address).Font.Size
    mixed = IsNull(fs) Or IsNull(area.Font.Size)
    If mixed Then
        size = topCell.Characters(1, 1).Font.Size
    Else
        size = fs
        If area.Font.Size <> size Then area.Font.Size = size
    End If

    v = topCell.Value
    If IsError(v) Then Exit Function
    txt = CStr(v)
    If txt = "" Then Exit Function

    ' 一時シートの A1 を同じ幅・同じフォントにして、行の高さを自動調整して測る
    Dim w As Double, i As Long
    For i = 1 To area.Columns.Count
        w = w + area.Columns(i).ColumnWidth
    Next i
    If w > 255 Then w = 255
    tmp.Columns(1).ColumnWidth = w
    With tmp.Range("A1")
        .NumberFormat = "@"
        .Value = txt
        .WrapText = True
        .Font.Name = topCell.Characters(1, 1).Font.Name      ' 先頭の文字の書式（Null にならない）
        .Font.Bold = topCell.Characters(1, 1).Font.Bold
        .Font.Italic = topCell.Characters(1, 1).Font.Italic
    End With

    Dim h As Double
    Do
        ' フォントを変えるたびに文字を入れ直し、行を低くしてから自動調整する
        tmp.Range("A1").ClearContents
        tmp.Range("A1").Font.Size = size
        tmp.Range("A1").Value = txt
        tmp.Rows(1).RowHeight = 12
        tmp.Rows(1).AutoFit
        h = tmp.Rows(1).RowHeight * HEIGHT_MARGIN + 2
        ' 余裕を含めて上限（から1ポイント以内）に届くなら入りきっていない → 小さくする
        ' （書式が混ざったセルは縮小しない）
        If mixed Or h < MAX_ROW_HEIGHT - 1 Or size <= MIN_FONT_SIZE Then Exit Do
        size = size - 0.5
    Loop
    If Not mixed Then
        If area.Font.Size <> size Then area.Font.Size = size
    End If
    FitOneCell = h
End Function

Private Function LoadSavedPath() As String
    LoadSavedPath = LoadText(NAME_SRC_PATH)
End Function

Private Sub SaveSourcePath(ByVal path As String)
    SaveText NAME_SRC_PATH, path
End Sub

' ブックの非表示の「名前」に文字列を保存する（空文字なら削除）
Private Sub SaveText(ByVal key As String, ByVal value As String)
    On Error Resume Next
    ThisWorkbook.Names(key).Delete
    On Error GoTo 0
    If value = "" Then Exit Sub
    ThisWorkbook.Names.Add Name:=key, _
        RefersTo:="=""" & Replace(value, """", """""") & """", Visible:=False
End Sub

Private Function LoadText(ByVal key As String) As String
    Dim s As String
    On Error Resume Next
    s = ThisWorkbook.Names(key).RefersTo
    If Err.Number <> 0 Then Exit Function
    On Error GoTo 0
    ' RefersTo は ="..." の形
    If Left$(s, 2) = "=""" Then s = Mid$(s, 3, Len(s) - 3)
    LoadText = Replace(s, """""", """")
End Function
